    standard PIBT procedure:
    C ← Neigh(πi[t]) ∪ {πi [t]}
    sort C in increasing order of dist (u, gi) where u ∈ C  //tie-break: presence of agents
    for v ∈ C do
        if ∃ak ∈ A s.t. πk[t + 1] = v then continue //avoid vertex conflict
        if aj 6= ⊥ ∧ πj [t] = v then continue //avoid swap conflict
        πi[t + 1] ← v  // reserve
        if ∃ak ∈ A s.t. πk[t] = v ∧ πk[t + 1] = ⊥ then
            if PIBT(ak, ai) is invalid then continue  // replanning
        end if
        return valid
    end for
    πi [t + 1] ← πi[t]
    return invalid


function compute_priority_chain(agent_i, cur_location, agents, frustrations,heuristic_fn, cache=[]):
    if len(cache) == 0: 
        cache = [0]*n 
    chain_frustration = 0 
    for neighbour in neighbouring cells:
        if neighbouring cell is occupied by an agent (agent_j): 
            if heuristic_fn(neighbouring_cell, goals[agent_j]) > heuristic_fn(cur_location, goals[agent_j]):  #should this be >=?
                neighbour_frustration = compute_priority_chain(agent_j, neighbouring_cell, agents, frustrations, cache)
                chain_frustration = max(chain_frustration, neighbour_frustration)
    return some_function(frustrations[agent_i], heuristic_fn(agent_i,goals[agent_i]) + chain_frustration
